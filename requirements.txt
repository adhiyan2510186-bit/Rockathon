# requirements.txt — every library this project needs, and nothing more.
#
# Install with:   python -m pip install -r requirements.txt
#
# Five libraries. We can defend each one to a judge in a sentence, which is the
# point — no dependency is here "just in case".

# The four screens (chat intake, comparison, approval, audit) are one Streamlit app.
# CLAUDE.md, "Tech stack": Interface is Streamlit. No FastAPI, no servers.
streamlit>=1.40

# Pydantic gives us one Product shape that both vendor catalogs get squeezed into,
# and it complains loudly if a field is missing or the wrong type. That is what
# makes the normaliser in discovery.py do real work instead of hoping.
pydantic>=2.9

# Reads config.yaml — the single file holding the authorisation limit, the per-unit
# cap defaults, the category weights and the 5-point substitution threshold.
# CLAUDE.md: limits live in a config file, never in code and never in a prompt.
PyYAML>=6.0

# The ONLY AI dependency in the project. Used by agent/language.py and nowhere else,
# for stages 0-2 (understanding the human's sentence). It never scores, ranks,
# or approves anything. Swapping providers later touches that one file.
google-genai>=1.0

# Runs the golden ranker test that asserts 58.0 / 48.7 / 33.7.
pytest>=8.0
